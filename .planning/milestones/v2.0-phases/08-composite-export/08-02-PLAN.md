  const check = checkWebCodecsSupport();
  setWebCodecsSupported(check.supported);
  if (!check.supported) setWebCodecsReason(check.reason);

handleExport callback:
1. Set state to 'preparing', progress to 0, clear errorMessage
2. Derive AudioConfig from mutedTracks (same logic as before -- all muted = 'none', else 'mix' with unmuted indices)
3. Get resolution config from EXPORT_RESOLUTIONS[resolution]
4. Set state to 'encoding'
5. Call `startExport()` with the command object and callbacks:
   ```typescript
   startExport(
     {
       type: 'start',
       files: results.map(r => r.originalFile),
       offsets: results.map(r => r.offsetSeconds),
       resolution: { width: res.width, height: res.height },
       fps: 30,
       bitrate: res.bitrate,
       audioConfig,
       totalDurationSeconds,
       tileAspectRatio,
     },
     {
       onProgress: (ratio) => setProgress(ratio),
       onComplete: (data) => {
         triggerDownload(new Uint8Array(data), 'composite.mp4', 'video/mp4');
         setExportState('complete');
         setTimeout(() => { setExportState('idle'); setProgress(0); }, 2000);
       },
       onError: (msg) => {
         setExportState('error');
         setErrorMessage(msg);
       },
       onCancelled: () => {
         setExportState('cancelled');
         setTimeout(() => { setExportState('idle'); setProgress(0); }, 1500);
       },
     },
   );
   ```

handleCancel callback:
```typescript
const handleCancel = useCallback(() => {
  cancelExport();
}, []);
```

handleRetry callback (same as before): reset to idle.
